\section{Tổng quan}
\subsection{Tính năng chính}\label{sec:1}


Main FSM là module duy nhất ra lệnh cho DMA và array. Mỗi phép GEMM được biểu diễn bằng một \textbf{job} (lấy từ bản chụp thanh ghi ở chế độ đơn, hoặc từ một descriptor ở chế độ chuỗi). Ba engine độc lập đưa job qua các pha: \textbf{Đọc} (FETCH, kiểm tra, LOAD\_*), \textbf{Tính} (COMPUTE) và \textbf{Ghi} (STORE\_C). Bộ chấm điểm bank quyết định khi nào một pha được bắt đầu, nên khi có ping-pong ba pha của ba job liên tiếp chạy chồng nhau mà không bao giờ đụng cùng một bank.



\paragraph{Chức năng}\label{tab:1-1}
\begin{itemize}
\item Chế độ đơn (START): một job, các pha CHECK\_CFG → LOAD\_BIAS (nếu \sig{POST_EN}) → LOAD\_A → LOAD\_B → COMPUTE → STORE\_C, tuần tự
\item Chế độ chuỗi (START\_CHAIN): mỗi descriptor là một job, chạy FETCH → kiểm tra → LOAD\_BIAS → LOAD\_A → LOAD\_B (trừ khi \sig{REUSE_B}) → COMPUTE → STORE\_C ; chuỗi dừng ở descriptor có \sig{LAST} = 1
\item Với \sig{HAS_PINGPONG} = 1, ba engine có thể chạy đồng thời khi các điều kiện bank và thứ tự job bên dưới cho phép. Với \sig{HAS_PINGPONG} = 0, mọi pha dùng bus hoặc scratchpad loại trừ nhau.
\item Nội dung DRAM sau một tác vụ DONE giống hệt thực thi tuần tự từng job, với mọi cấu hình tham số và mọi thời gian trả lời của bus
\item FSM báo kết thúc tác vụ cho Register File bằng đúng một xung done hoặc một xung error; điều phối soft reset; cấp bộ đếm \sig{DESC_DONE_CNT}, thanh ghi \sig{ERR_DESC_ADDR} và tín hiệu đếm hiệu năng
\item FSM cấp cấu hình của job đang tính cho array và VPU, và bank đang tính cho scratchpad. Bản chụp của Register File chỉ đi tới FSM
\end{itemize}

\end{spectable}

\end{spectable}


\paragraph{Giao tiếp}\label{tab:3-2}
\begin{itemize}
\item Mọi port \sig{o_*} lấy từ thanh ghi, trừ \sig{o_rd_abort} và \sig{o_wr_abort} (mức, cũng từ thanh ghi trạng thái)
\item Lệnh DMA: FSM chỉ bật \sig{o_rd_start} khi kênh đọc rảnh (\sig{i_rd_busy} = 0 và không có lệnh nào đang chờ \sig{i_rd_done}/\sig{i_rd_error}); tương tự kênh ghi. Các port tham số chỉ cần hợp lệ ở chu kỳ \sig{o_*_start} = 1; DMA tự chốt
\item \sig{o_comp_dim}, \sig{o_comp_post_en}, \sig{o_comp_relu_en}, \sig{o_comp_qm0}, \sig{o_comp_qshift}, \sig{o_comp_bank_*} hợp lệ từ chu kỳ \sig{o_pe_start} = 1 tới hết chu kỳ \sig{i_pe_done} = 1 và không đổi trong khoảng đó
\item \sig{o_pe_start} chỉ khi \sig{i_pe_busy} = 0. Mỗi \sig{o_pe_start} có đúng một \sig{i_pe_done}
\item \sig{o_done_pulse} và \sig{o_err_pulse} không bao giờ cùng chu kỳ; mỗi \sig{i_start_pulse} có tối đa một trong hai; không có cái nào cho tác vụ bị soft reset hủy
\end{itemize}

\paragraph{Kiểm tra descriptor (\sigc{gemm_job_check}, mode = chuỗi)}\label{tab:5-3}
\begin{itemize}
\item Hợp lệ khi và chỉ khi mọi điều sau đúng : $1 ≤ M ≤ 32$; N, K ∈ \{4, 8, …, 32\}; w0[31:24] = 0; w5[31:16] = 0; w6[31:5] = 0; w10[31:5] = 0; w12 … w15 = 0
\item \sig{a}[1:0] = 0 và \sig{c}[1:0] = 0; \sig{b}[1:0] = 0 nếu \sig{reuse} = 0; \sig{bias}[1:0] = 0 nếu \sig{post} = 1
\item Mỗi stride thô (w4[15:0], w4[31:16], w5[15:0]) bằng 0, hoặc là bội của 4 và không nhỏ hơn độ dài hàng: K cho A; N cho B; 4N cho C khi \sig{post} = 0, N khi \sig{post} = 1
\item \sig{relu} = 1 kéo theo \sig{post} = 1
\item \sig{reuse} = 1 chỉ khi \sig{idx} > 0 và (K, N) bằng (K, N) của job trước trong chuỗi (thanh ghi \sig{prev_kn} của engine Đọc)
\item \sig{last} = 0 kéo theo \sig{next}[5:0] = 0 . Vi phạm được tính là lỗi của chính descriptor này
\item \sig{DESC_PTR}[5:0] ≠ 0 được kiểm ở \sig{gemm_fsm_ctrl} khi bắt đầu chuỗi: lỗi mã 1 với f = 0, \sig{ERR_DESC_ADDR} = \sig{DESC_PTR}, không truy cập bus
\end{itemize}

\subsubsection{Chọn bank}\label{sec:6.1}

\paragraph{\sigc{gemm_bank_sb}: chọn bank }\label{tab:6-1}

\begin{spectable}{|L{0.17}|L{0.12}|L{0.36}|L{0.35}|}{Bộ đếm của \sigc{gemm_bank_sb}}{tab:6-2}{\textbf{Bộ đếm} & \textbf{Rộng} & \textbf{Tăng khi} & \textbf{Giảm khi}}

\sig{rcnt[B][b]} & 2 & LOAD\_B vào bank b, hoặc job \sig{reuse} = 1 có \sig{bb} = b đi qua bước B & COMPUTE của job có \sig{bb} = b xong \\\hline

\paragraph{Điều kiện bank }\label{tab:6-3}
\begin{itemize}
\item LOAD\_X vào bank b chỉ được bắt đầu khi \sig{rcnt[X][b]} = 0: mọi job đọc bank đó đã tính xong. Job \sig{reuse} = 1 không chờ (dữ liệu B của bank đó vẫn còn, vì chưa có LOAD\_B nào vào bank ấy kể từ job trước)
\item COMPUTE vào bank C b chỉ được bắt đầu khi \sig{wcnt[C][b]} = 0: STORE\_C cuối cùng đọc bank đó đã xong
\item Bộ đếm không bao giờ tràn hay âm (assertion). \sig{rcnt} tối đa 3 vì tối đa ba job đã gán bank mà chưa tính xong (engine Đọc, slot, engine Tính)
\item Khi tác vụ kết thúc do lỗi, bộ đếm có thể khác 0; chúng được xóa khi chấp nhận tác vụ kế tiếp hoặc khi \sig{soft_clr}
\end{itemize}


\subsubsection{Cấp quyền bắt đầu pha}\label{sec:6.3}
Mỗi engine tính tín hiệu \sig{want} từ trạng thái đầu chu kỳ: \sig{want_s} (Ghi), \sig{want_c} (Tính), \sig{want_r} (Đọc: FETCH hoặc một LOAD). \sig{gemm_fsm_ctrl} trả \sig{go}.

\eqbox{\[
\begin{aligned}
&\text{PP } (NB = 2): && go\_x = want\_x \wedge \neg inh \\
&\text{SEQ } (NB = 1): && go\_s = want\_s \wedge \neg act \wedge \neg inh,\quad go\_c = want\_c \wedge \neg act \wedge \neg want\_s \wedge \neg inh,\\
& && go\_r = want\_r \wedge \neg act \wedge \neg want\_s \wedge \neg want\_c \wedge \neg inh
\end{aligned}
\]}
Trong đó \sig{act} = (engine Đọc đang chờ một lệnh DMA) ∨ (engine Tính ở RUN) ∨ (engine Ghi ở RUN); \sig{inh} = tác vụ không ở trạng thái RUN (gồm SOFT, CLR). Ưu tiên Ghi > Tính > Đọc làm job cũ ra trước và bảo đảm lịch tuần tự.


\paragraph{Chặn sau lỗi }\label{tab:6-4}
\begin{itemize}
\item \sig{gemm_fsm_ctrl} cấp \sig{blk_v}, \sig{blk_idx} tổ hợp: bản ghi lỗi hiện có, gộp với lỗi đến trong chính chu kỳ này (\sig{i_rd_error}, \sig{i_wr_error}, kết quả CHECK sai), lấy chỉ số nhỏ nhất
\item Mọi \sig{want} của job có \sig{idx} ≥ \sig{blk_idx} bằng 0 khi \sig{blk_v} = 1. Nhờ gộp lỗi cùng chu kỳ, không pha nào của descriptor ≥ f bắt đầu, kể cả ở đúng chu kỳ lỗi được báo
\item Pha đã bắt đầu thì chạy tới hết (không abort khi lỗi). Kết quả của chúng không được ghi ra DRAM vì STORE\_C của job ≥ f không bao giờ bắt đầu
\end{itemize}

\begin{notebox}[Vì sao engine Tính giữ job ở C\_HAND không làm chậm]
COMPUTE(d+1) dùng bank C của job d−1, nên theo (COMPUTE(d) bắt đầu khi mọi LOAD của d và COMPUTE(d−1) đã xong, và STORE\_C cuối cùng đọc bank C của d đã xong ) chỉ bắt đầu được khi STORE\_C(d−1) xong. Đúng lúc đó engine Ghi rảnh và nhận job d, nên engine Tính về C\_IDLE và có thể bắt đầu d+1 ngay chu kỳ sau. Với NB = 1, lịch vốn đã tuần tự.
\end{notebox}

T\_IDLE & Chờ. \sig{i_start_pulse}: xóa bản ghi lỗi, \sig{desc_done_cnt}, \sig{err_desc_addr}, con trỏ bank, bộ đếm; xét \sig{i_start_kind}  & T\_RUN. \sig{i_soft_req}: T\_CLR \\\hline

T\_RUN & Engine chạy. Kiểm điều kiện kết thúc  & T\_IDLE khi done/err; T\_SOFT khi \sig{i_soft_req} \\\hline

T\_SOFT & \sig{inh} = 1; bật \sig{o_rd_abort} khi kênh đọc bận, \sig{o_wr_abort} khi kênh ghi bận; không báo done/err & T\_CLR khi không engine nào chờ lệnh DMA \\\hline

T\_CLR & \sig{o_soft_clr} = 1 trong 1 chu kỳ; mọi thanh ghi của FSM về giá trị reset & T\_IDLE \\\hline

\end{spectable}

\begin{spectable}{|L{0.2}|L{0.8}|}{Xử lý \sigc{i_start_kind}}{tab:8-2}{\textbf{Giá trị} & \textbf{Hành vi}}
2'b01 (START) & Chế độ đơn: engine Đọc vào R\_START \\\hline

2'b10 (START\_CHAIN) & \sig{HAS_CHAIN} = 0: lỗi mã 5. \sig{DESC_PTR}[5:0] ≠ 0: lỗi mã 1 . Ngược lại: chế độ chuỗi \\\hline

2'b11 & Lỗi mã 1 . Không engine nào chạy, không truy cập bus \\\hline

\end{spectable}


\paragraph{Kết thúc tác vụ}\label{tab:8-3}
\begin{itemize}
\item Bản ghi lỗi (\sig{err_v}, \sig{err_idx}, \sig{err_code}, \sig{err_addr}) chỉ được thay khi lỗi mới có idx nhỏ hơn . Mã: FETCH → 4, CHECK → 1, LOAD\_* → 2, STORE\_C → 3. \sig{err_addr} = \sig{daddr} của job lỗi; = con trỏ đang FETCH với lỗi mã 4; = \sig{DESC_PTR} với (\sig{DESC_PTR}[5:0] ≠ 0 được kiểm ở \sig{gemm_fsm_ctrl} khi bắt đầu chuỗi: lỗi mã 1 với f = 0, \sig{ERR_DESC_ADDR} = \sig{DESC_PTR}, không truy cập bus )
\item DONE: ở T\_RUN, engine Ghi báo job có \sig{last} = 1 xong OKAY và \sig{err_v} = 0 → \sig{o_done_pulse} ở chu kỳ kế tiếp
\item ERROR: ở T\_RUN, \sig{err_v} = 1 và hệ yên: engine Đọc ở R\_STOP và không chờ DMA; engine Tính không ở C\_RUN; slot trống hoặc có idx ≥ f; engine Tính không giữ ở C\_HAND một job idx < f; engine Ghi ở S\_IDLE → \sig{o_err_pulse} với \sig{o_err_code} = \sig{err_code} ở chu kỳ kế tiếp; \sig{o_err_desc_addr} ← \sig{err_addr} (chỉ ở chế độ chuỗi)
\item Khi đó \sig{o_desc_done_cnt} = f tự nhiên, vì mọi job < f đã ghi xong và không job ≥ f nào được ghi
\item \sig{o_err_desc_addr} và \sig{o_desc_done_cnt} bằng 0 ở chế độ đơn và khi \sig{HAS_CHAIN} = 0
\end{itemize}



\paragraph{Soft reset }\label{tab:8-4}
\begin{itemize}
\item \sig{o_soft_busy} = 1 ở T\_SOFT và T\_CLR
\item Ở T\_SOFT: không pha mới nào bắt đầu; lệnh DMA đang chạy kết thúc bằng done hoặc error như thường ; engine vẫn nhận các xung đó để rời trạng thái chờ, nhưng không cập nhật bản ghi lỗi hay báo done
\item COMPUTE đang chạy không được chờ: \sig{o_soft_clr} hủy array và VPU
\item \sig{i_soft_req} ở T\_IDLE: T\_CLR ở chu kỳ kế tiếp. \sig{i_soft_req} ở T\_SOFT hoặc T\_CLR: bỏ qua
\item Nếu \sig{i_soft_req} đến cùng chu kỳ với \sig{i_rd_done} hoặc \sig{i_wr_done}, kênh đó coi như đã rảnh
\end{itemize}

\subsection{Reset, hiện thực, kiểm chứng}\label{sec:10}


\paragraph{Reset}\label{tab:10-1}
\begin{itemize}
\item \sig{rst_n} = 0 hoặc T\_CLR: mọi engine về trạng thái IDLE, slot trống, bản ghi lỗi, con trỏ bank, bộ đếm, \sig{desc_done_cnt}, \sig{err_desc_addr}, mọi xung đầu ra về 0. Trường dữ liệu của job không bắt buộc reset
\end{itemize}



\paragraph{Hiện thực}\label{tab:10-2}
\begin{itemize}
\item SystemVerilog tổng hợp được; FSM viết theo kiểu hai khối (\sig{always_ff} trạng thái, \sig{always_comb} trạng thái kế); không có latch
\item \sig{job_t} khai báo trong \sig{gemm_pkg} là \sig{struct packed}; công cụ tổng hợp tự bỏ trường không dùng ở mỗi engine
\item Với \sig{HAS_CHAIN} = 0, logic FETCH, bộ đệm và kiểm tra descriptor, cùng \sig{desc_done_cnt} được loại bằng \sig{generate}. Với NB = 1, bộ đếm và con trỏ bank được loại, các điều kiện bank luôn đúng trừ quy tắc SEQ
\item Mọi so sánh idx dùng 32 bit không dấu
\end{itemize}



\paragraph{Assertion bắt buộc (SVA)}\label{tab:10-3}
\begin{itemize}
\item \sig{o_rd_start} → kênh đọc rảnh; \sig{o_wr_start} → kênh ghi rảnh; \sig{o_pe_start} → \sig{!i_pe_busy}
\item NB = 1: tại mọi chu kỳ có tối đa một trong \{engine Đọc chờ DMA, C\_RUN, S\_RUN\}
\item Không bộ đếm nào tràn hay âm
\item LOAD\_X bắt đầu vào bank b → không job nào ở slot, C\_RUN hay C\_HAND chưa tính xong có bank X = b (trừ job reuse)
\item STORE\_C bắt đầu theo thứ tự idx tăng dần 1
\item \sig{o_done_pulse} ∧ \sig{o_err_pulse} không bao giờ cùng lúc; mỗi \sig{i_start_pulse} có tối đa một trong hai
\item \sig{o_comp_*} ổn định trong khi C\_RUN
\end{itemize}
