\section{Tổng quan}
\subsection{Tính năng chính}\label{sec:1}


Scratchpad giữ bốn buffer logic A, B, C, BIAS, mỗi buffer có NB bank. Phía tính (array, VPU) cần đọc hoặc ghi P phần tử trong một chu kỳ, nên mỗi bank được chia thành nhiều \textbf{sub-bank} một cổng sao cho mỗi truy cập song song chạm các sub-bank khác nhau. Phía DMA đọc hoặc ghi một word mỗi chu kỳ. Hai phía dùng hai bank khác nhau (ping-pong) hoặc hai pha khác nhau (không ping-pong), nên không bao giờ tranh nhau một sub-bank.



\paragraph{Chức năng}\label{tab:1-1}
\begin{itemize}
\item Mỗi bank có dung lượng: A 256 word, B 256 word, C 1024 word, BIAS 32 word; tổng NB × 1568 word (NB = 2: 12,25 KiB)
\item Mọi cổng đọc có độ trễ cố định 1 chu kỳ; mọi lệnh ghi có hiệu lực ở cạnh clock cuối chu kỳ có lệnh; không có backpressure
\item Mỗi sub-bank vật lý là RAM một cổng đồng bộ. Bank b của buffer X chỉ bị truy cập bởi một phía trong một chu kỳ ; hai bank khác nhau được truy cập độc lập cùng chu kỳ
\item Nội dung không được khởi tạo bởi \sig{rst_n} hay \sig{soft_clr}
\end{itemize}

\subsection{Điều kiện phục vụ đồng thời}
Trong một chu kỳ SRAM phải phục vụ đồng thời một lần đọc A (P byte từ P hàng), một lần đọc B (P byte liền nhau), một lần đọc BIAS (P word) và một lần ghi C (P phần tử có mask). Hai phía tính và DMA được chạy đồng thời trên các bank khác nhau. Khi NB = 1, môi trường chỉ cho phép một phía hoạt động tại một thời điểm.
Đọc A/B/BIAS trả dữ liệu ở chu kỳ kế tiếp; bên nhận chỉ dùng các lane hợp lệ. Ghi C với fmt = 0 lấy phần tử c từ data[32c +: 32] và mask[c] cho phép ghi cả word. Với fmt = 1, lấy byte c từ data[8c +: 8], mask[c] cho phép ghi byte đó; phần cao của data bằng 0. Hàng/cột đầu phải hợp lệ và cột đầu là bội của P. Mask chặn các phần tử ngoài miền tác vụ.
Mỗi yêu cầu đọc DMA được nhận phải tạo đúng một rvalid ở chu kỳ tiếp theo, theo đúng thứ tự; không có ready cho dữ liệu trả về. Nội dung RAM và các lane không hợp lệ không cần khởi tạo, và được phép chứa X trong mô phỏng.

\subsubsection{Truy cập phía tính}

\paragraph{Cổng phía tính (b = \sigc{i_comp_bank_*})}\label{tab:5-3}
\begin{itemize}
\item Đọc A ($i_0$, k): sub-bank r (0 ≤ r < P) của A[b] đọc địa chỉ \{$i_0$[4:LP], k[4:2]\}. Lane r = byte k[1:0] của sub-bank r; k[1:0] được chốt để chọn byte ở chu kỳ sau
\item Đọc B (k, $j_0$): sub-bank s (0 ≤ s < Q) của B[b] đọc \{k, ($j_0$ » 2) » log$_2$Q\}. Lane c = byte (c mod 4) của sub-bank c/4
\item Đọc BIAS ($j_0$): sub-bank c của BIAS[b] đọc $j_0$ » LP; lane c = sub-bank c
\item Ghi C INT32 (i, $j_0$): lane c ghi sub-bank c của C[b] tại (\{i, $j_0$\} » LP), cả 4 byte, nếu \sig{mask[c]}
\item Ghi C INT8 (i, $j_0$): word thứ w (0 ≤ w < Q) của đoạn hàng có địa chỉ a = \{i, $j_0$/4 + w\}, nằm ở sub-bank s = ($j_0$/4 + w) mod P, địa chỉ a » LP. Sub-bank s nhận word w = (s − $j_0$/4) mod P nếu w < Q; byte e của nó = \sig{i_c_wr_data[8(4w+e) +: 8]}, byte enable e = \sig{mask[4w+e]}
\end{itemize}


\subsubsection{Truy cập phía DMA}

\paragraph{Cổng DMA}\label{tab:5-4}
\begin{itemize}
\item Ghi \sig{i_dw_*}: sub-bank và địa chỉ cục bộ theo bảng Sub-bank của mỗi bank với a = \sig{i_dw_addr}, trong buffer \sig{i_dw_buf}, bank \sig{i_dw_bank}; cả 4 byte
\item Đọc \sig{i_dr_*}: sub-bank (a mod P) của C[\sig{i_dr_bank}], địa chỉ a » LP. Số sub-bank và bank được chốt để chọn dữ liệu ra ở chu kỳ sau
\end{itemize}


\begin{notebox}[Vì sao không bao giờ xung đột]
$i_0$ và $j_0$ là bội của P, nên P hàng của một đọc A rơi vào P sub-bank khác nhau, P cột của một đọc BIAS hoặc ghi C INT32 cũng vậy; Q word của một đọc B hoặc ghi C INT8 nằm liền nhau nên rơi vào Q sub-bank khác nhau. Ở INT8, \{i, jw\} = 8i + jw và 8i chia hết cho P, nên sub-bank chỉ phụ thuộc jw. Mô hình kiểm đủ mọi (i$_0$, k), (k, j$_0$), mọi đoạn hàng C, với P = 4, 8.
\end{notebox}

\subsubsection{Chọn bên truy cập}

\paragraph{Mux cổng của mỗi sub-bank (X, b, s)}\label{tab:5-5}
\begin{itemize}
\item \sig{dma_sel} = lệnh DMA (ghi \sig{i_dw_*} cho A, B, BIAS; đọc \sig{i_dr_*} cho C) có buffer X, bank b và sub-bank s
\item \sig{cmp_sel} = lệnh phía tính tương ứng (\sig{i_a_rd_en}, \sig{i_b_rd_en}, \sig{i_bias_rd_en}, \sig{i_c_wr_en} với byte enable khác 0 cho s) và \sig{i_comp_bank_X} = b
\item RAM: \sig{en = dma_sel ∨ cmp_sel}; địa chỉ, dữ liệu, byte enable lấy từ bên được chọn. Nếu cả hai , DMA thắng
\item Không cần tín hiệu pha: quyền truy cập do bank và lịch của FSM bảo đảm
\end{itemize}

\sig{o_rdata} & 32 & Output & Word 32 bit đọc từ RAM.\newline Hợp lệ một chu kỳ sau lệnh có i\_en = 1 và i\_we = 0. \\\hline

\end{spectable}

\begin{spectable}{|L{0.2}|L{0.8}|}{Module con theo buffer}{tab:6-2}{\textbf{Module} & \textbf{Nội dung}}
\sig{gemm_spm_a} & NB × P RAM; phục vụ đọc vector A và ghi word DMA; chốt k[1:0] và bank để chọn byte trả về. \\\hline

\sig{gemm_spm_b} & NB × Q RAM; phục vụ đọc vector B và ghi word DMA; chốt bank để chọn dữ liệu trả về. \\\hline

\sig{gemm_spm_bias} & NB × P RAM; dùng thanh ghi vì mỗi RAM chỉ sâu 4 hoặc 8 word; đọc vector bias và ghi word DMA. \\\hline

\sig{gemm_spm_c} & NB × P RAM có byte enable; ghi đoạn hàng INT32/INT8 và đọc word DMA; xoay word khi ghi INT8, chốt bank và sub-bank để chọn dữ liệu đọc. \\\hline

\end{spectable}

\begin{Verbatim}[frame=leftline,framerule=2pt,rulecolor=\color{brand},xleftmargin=6pt,samepage=true]
// gemm_spm_ram: RAM một cổng, đọc đồng bộ (tham khảo)
logic [31:0] mem [DEPTH];
always_ff @(posedge clk)
  if (i_en) begin
    if (i_we) begin
      for (int e = 0; e < 4; e++)
        if (!BE || i_be[e]) mem[i_addr][8*e +: 8] <= i_wdata[8*e +: 8];
    end else
      o_rdata <= mem[i_addr];
  end
\end{Verbatim}

\subsection{Reset, hiện thực, kiểm chứng}\label{sec:8}

\paragraph{Reset}\label{tab:8-1}
\begin{itemize}
\item \sig{rst_n} = 0 hoặc \sig{i_soft_clr}: \sig{o_dr_rvalid} về 0. Thanh ghi chọn byte, bank, sub-bank và nội dung RAM không bị reset
\end{itemize}



\paragraph{Hiện thực}\label{tab:8-2}
\begin{itemize}
\item RAM viết theo mẫu \sig{gemm_spm_ram} để công cụ suy ra block RAM hoặc LUTRAM có đọc đồng bộ. Không được dùng RAM đọc bất đồng bộ cho A, B, C
\item Không có mux tổ hợp từ \sig{*_rd_en} tới dữ liệu ra; đường ra là: thanh ghi đầu ra RAM → mux bank → mux byte/lane
\item Mỗi buffer trong file riêng; NB và số sub-bank dựng bằng \sig{generate}
\end{itemize}



\paragraph{Assertion bắt buộc (SVA)}\label{tab:8-3}
\begin{itemize}
\item \sig{i_dw_valid} → \sig{i_dw_buf} ≠ 3 ∧ địa chỉ trong miền của buffer (A, B < 256; BIAS < 32)
\item NB = 2: \sig{i_dw_valid} ∧ \sig{i_comp_active} → \sig{i_dw_bank} ≠ \sig{i_comp_bank_X} với X = \sig{i_dw_buf}; và \sig{i_dr_valid} ∧ \sig{i_comp_active} → \sig{i_dr_bank} ≠ \sig{i_comp_bank_c}. NB = 1: \sig{i_dw_valid} ∨ \sig{i_dr_valid} → ¬\sig{i_comp_active}
\item \sig{i_a_rd_en} ∨ \sig{i_b_rd_en} ∨ \sig{i_bias_rd_en} ∨ \sig{i_c_wr_en} → \sig{i_comp_active}
\item Không sub-bank nào có \sig{dma_sel} ∧ \sig{cmp_sel} cùng chu kỳ
\item \sig{i_a_rd_row}, \sig{i_b_rd_col}, \sig{i_bias_rd_col}, \sig{i_c_wr_col} là bội của P khi có lệnh
\end{itemize}
