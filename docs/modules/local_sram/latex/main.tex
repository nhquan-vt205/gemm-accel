\section{Tổng quan}
\subsection{Tính năng chính}\label{sec:1}

Scratchpad giữ bốn buffer logic A, B, C, BIAS, mỗi buffer có hai bank (ping-pong, NB = 2). Phía tính (array, VPU) cần đọc hoặc ghi 8 phần tử trong một chu kỳ, nên mỗi bank được chia thành nhiều \textbf{sub-bank} một cổng sao cho mỗi truy cập song song chạm các sub-bank khác nhau. Phía DMA đọc hoặc ghi một word mỗi chu kỳ. Hai phía dùng hai bank khác nhau, nên không bao giờ tranh nhau một sub-bank.

\paragraph{Chức năng}
\begin{itemize}
\item Mỗi buffer có hai bank. Mỗi bank: A 256 word, B 256 word, C 1024 word, BIAS 32 word, tức 1568 word mỗi bank; tổng 3136 word = 12,25 KiB
\item Mọi cổng đọc có độ trễ cố định đúng 1 chu kỳ; mọi lệnh ghi có hiệu lực ở cạnh clock cuối chu kỳ có lệnh; không có backpressure (\sig{o_dw_ready} = \sig{o_dr_ready} = 1 hằng số)
\item Mỗi sub-bank là SRAM một cổng đồng bộ. Bank b của buffer X chỉ bị truy cập bởi một phía trong một chu kỳ; hai bank khác nhau được truy cập độc lập cùng chu kỳ
\item Mỗi chu kỳ phía tính truy cập đồng thời A, B, BIAS, C; phía DMA truy cập đồng thời một cổng ghi (A, B hoặc BIAS) và một cổng đọc (C)
\item Nội dung RAM không được khởi tạo bởi \sig{rst_n} hay \sig{soft_clr}; nội dung sau reset không xác định
\end{itemize}

\paragraph{Giao tiếp}
\begin{itemize}
\item \sig{o_dw_ready} và \sig{o_dr_ready} luôn bằng 1. Lệnh chỉ được nhận khi valid và ready cùng bằng 1. Mỗi chu kỳ nhận một word ghi và một yêu cầu đọc; mỗi yêu cầu đọc có đúng một phản hồi theo thứ tự
\item \sig{o_a_rd_data}, \sig{o_b_rd_data}, \sig{o_bias_rd_data} hợp lệ ở đúng chu kỳ sau chu kỳ có \sig{*_rd_en}; ở chu kỳ khác giá trị tùy ý. \sig{o_dr_rvalid} là thanh ghi
\item Lane ứng với hàng hoặc cột ngoài vùng hợp lệ của tác vụ mang nội dung tùy ý
\end{itemize}

\subsection{Điều kiện phục vụ đồng thời}
Trong một chu kỳ SRAM phải phục vụ đồng thời một lần đọc A (8 byte từ 8 hàng), một lần đọc B (8 byte liền nhau), một lần đọc BIAS (8 word) và một lần ghi C (8 phần tử có mask). Hai phía tính và DMA chạy đồng thời trên các bank khác nhau.

\begin{spectable}{|L{0.2}|L{0.2}|L{0.6}|}{Truy cập song song và sub-bank bị chạm (P = 8)}{tab:3-2}{\textbf{Truy cập} & \textbf{Sub-bank chạm} & \textbf{Vì sao không trùng}}
Đọc A (\sig{i_a_rd_*}) & 8 của A[b] & $i_0$ là bội của 8, nên 8 hàng $i_0 + r$ có $i \bmod 8$ khác nhau \\\hline
Đọc B (\sig{i_b_rd_*}) & 2 của B[b] & $j_0$ là bội của 8, nên hai word $jw = j_0/4,\ j_0/4+1$ có $jw \bmod 2$ khác nhau \\\hline
Đọc BIAS (\sig{i_bias_rd_*}) & 8 của BIAS[b] & 8 cột $j_0 + c$ có $j \bmod 8$ khác nhau \\\hline
Ghi C INT32 (\sig{i_c_wr_*}, fmt = 0) & 8 của C[b] & 8 cột $j_0 + c$ có $j \bmod 8$ khác nhau \\\hline
Ghi C INT8 (fmt = 1) & 2 của C[b] & hai word liền nhau $jw$ có $jw \bmod 8$ khác nhau \\\hline
DMA ghi A, B, BIAS & 1 & một word mỗi chu kỳ \\\hline
DMA đọc C & 1 & một word mỗi chu kỳ \\\hline

\end{spectable}

\begin{itemize}
\item Phía tính chạm bank \sig{i_comp_bank_*} của cả bốn buffer; phía DMA chạm buffer và bank do \sig{i_dw_*}, \sig{i_dr_*} chọn. Các buffer khác nhau là các RAM khác nhau nên không tranh nhau; trong cùng một buffer, Main FSM cấp hai bank khác nhau cho hai phía (scoreboard) khi phía tính thực sự dùng buffer đó, được kiểm bằng AS-02. Riêng BIAS: job có \sig{POST_EN} = 0 vẫn mang một giá trị \sig{i_comp_bank_bias} nhưng không đọc BIAS, nên DMA được phép ghi BIAS vào bank đó
\item Đọc A/B/BIAS trả dữ liệu ở chu kỳ kế tiếp; bên nhận chỉ dùng các lane hợp lệ. Ghi C với fmt = 0 lấy phần tử c từ \sigc{data[32c +: 32]} và \sig{mask[c]} cho phép ghi cả word. Với fmt = 1, lấy byte c từ \sigc{data[8c +: 8]}, \sig{mask[c]} cho phép ghi byte đó; phần cao \sig{data[255:64]} bằng 0. Hàng/cột đầu phải hợp lệ và cột đầu là bội của 8. Mask chặn các phần tử ngoài miền tác vụ
\item Mỗi yêu cầu đọc DMA được nhận phải tạo đúng một \sig{o_dr_rvalid} ở chu kỳ tiếp theo, theo đúng thứ tự; không có ready cho dữ liệu trả về. Nội dung RAM và các lane không hợp lệ không cần khởi tạo, và được phép chứa X trong mô phỏng
\end{itemize}

\section{Đặc tả thanh ghi}
Module không có thanh ghi ánh xạ MMIO. Toàn bộ thanh ghi MMIO nằm trong \sig{gemm_register_file}, xem đặc tả hệ thống baseline. Cấu hình được nhận qua các tín hiệu đầu vào (bank phía tính từ Main FSM, lệnh từ DMA, array và VPU); trạng thái nội bộ, điều kiện chốt và reset được mô tả bên dưới.

\section{Mô tả chức năng}
\subsection{Bố cục logic}\label{sec:4}

\begin{spectable}{|L{0.12}|L{0.25}|L{0.63}|}{Địa chỉ word logic trong mỗi bank}{tab:4-1}{\textbf{Buffer} & \textbf{Địa chỉ a} & \textbf{Nội dung word}}
A & \{i[4:0], kw[2:0]\}, kw = k » 2 & Byte e = $A[i][4kw + e]$; lane = k[1:0] \\\hline

B & \{k[4:0], jw[2:0]\}, jw = j » 2 & Byte e = $B[k][4jw + e]$ \\\hline

BIAS & \{j[4:0]\} & $bias[j]$ \\\hline

C, INT32 & \{i[4:0], j[4:0]\} & $C[i][j]$ (word 0 … 1023) \\\hline

C, INT8 & \{i[4:0], jw[2:0]\} & Byte e = $Y[i][4jw + e]$, lane = j[1:0] (dùng word 0 … 255) \\\hline

\end{spectable}

\begin{itemize}
\item Mỗi phần tử A, B là INT8 có dấu; bốn phần tử liên tiếp theo K (A) hoặc theo N (B) nằm trong một word 32 bit, phần tử có chỉ số thấp ở byte thấp (little-endian, trùng thứ tự byte của DRAM)
\item Cùng một buffer C chứa INT32 (khi \sig{POST_EN} = 0) hoặc INT8 (khi \sig{POST_EN} = 1); định dạng do \sig{i_c_wr_fmt} chọn ở mỗi lệnh ghi và do \sig{i_wr_fmt} của DMA chọn khi đọc. SRAM không lưu định dạng
\item Quy tắc đóng gói INT8 phía VPU: phần tử c nằm ở \sig{i_c_wr_data[8c+7:8c]}, \sig{i_c_wr_data[255:64]} = 0; \sig{i_c_wr_mask[c]} là byte enable của đúng byte đó
\end{itemize}

\begin{notebox}[Quyết định baseline]
\textbf{D-28 (kích thước và độ trễ).} Mỗi bank: A 256 word, B 256 word, C 1024 word, BIAS 32 word, tổng 1568 word mỗi bank, 3136 word cho cả hai bank (12,25 KiB). Mỗi sub-bank là SRAM một cổng. Độ trễ đọc đúng 1 chu kỳ; ghi có hiệu lực ở cạnh clock cuối chu kỳ lệnh; không có backpressure, \sig{o_dw_ready} = \sig{o_dr_ready} = 1.
\end{notebox}

\subsubsection{Truy cập phía tính}

\paragraph{Cổng phía tính (b = \sigc{i_comp_bank_*})}
\begin{itemize}
\item Đọc A ($i_0$, k): sub-bank r (0 ≤ r < 8) của A[b] đọc địa chỉ \{$i_0$[4:3], k[4:2]\}. Lane r = byte k[1:0] của sub-bank r; k[1:0] được chốt để chọn byte ở chu kỳ sau
\item Đọc B (k, $j_0$): sub-bank s (s = 0, 1) của B[b] đọc \{k, $j_0$[4:3]\}. Lane c = byte (c mod 4) của sub-bank c/4
\item Đọc BIAS ($j_0$): sub-bank c (0 ≤ c < 8) của BIAS[b] đọc $j_0$[4:3]; lane c = sub-bank c
\item Ghi C INT32 (i, $j_0$): lane c ghi sub-bank c của C[b] tại \{i[4:0], $j_0$[4:3]\}, dữ liệu \sigc{i_c_wr_data[32c +: 32]}, cả 4 byte, nếu \sig{mask[c]}
\item Ghi C INT8 (i, $j_0$): word w (w = 0, 1) của đoạn hàng có địa chỉ a = \{i, $j_0$/4 + w\}, nằm ở sub-bank s = \{$j_0$[4:3], w\}, địa chỉ cục bộ a » 3 = i (chiếm 32 word đầu của sub-bank). Sub-bank s nhận word w = s[0] nếu s[2:1] = $j_0$[4:3]: dữ liệu \sigc{i_c_wr_data[32w +: 32]}, byte enable e = \sig{mask[4w+e]}. Các sub-bank còn lại không được chạm
\end{itemize}

\subsubsection{Truy cập phía DMA}

\begin{spectable}{|L{0.14}|L{0.22}|L{0.3}|L{0.34}|}{Ánh xạ cổng DMA (a = địa chỉ word logic)}{tab:5-3}{\textbf{Cổng} & \textbf{Buffer} & \textbf{Sub-bank} & \textbf{Địa chỉ cục bộ}}
Ghi \sig{i_dw_*} & A, \sig{i_dw_buf} = 0 & a[5:3] (= i[2:0]) & \{a[7:6], a[2:0]\} \\\hline
Ghi \sig{i_dw_*} & B, \sig{i_dw_buf} = 1 & a[0] (= jw[0]) & a[7:1] \\\hline
Ghi \sig{i_dw_*} & BIAS, \sig{i_dw_buf} = 2 & a[2:0] & a[4:3] \\\hline
Đọc \sig{i_dr_*} & C & a[2:0] & a[9:3] \\\hline

\end{spectable}

\begin{itemize}
\item Ghi \sig{i_dw_*}: buffer \sig{i_dw_buf}, bank \sig{i_dw_bank}, cả 4 byte của \sig{i_dw_data}; \sig{i_dw_buf} = 3 là không hợp lệ (AS-01)
\item Đọc \sig{i_dr_*}: C[\sig{i_dr_bank}], sub-bank và địa chỉ cục bộ theo bảng. Số sub-bank và bank được chốt để chọn dữ liệu ra ở chu kỳ sau
\end{itemize}

\begin{notebox}[Vì sao không bao giờ xung đột]
$i_0$ và $j_0$ là bội của 8, nên 8 hàng của một đọc A rơi vào 8 sub-bank khác nhau, 8 cột của một đọc BIAS hoặc ghi C INT32 cũng vậy; hai word của một đọc B hoặc ghi C INT8 nằm liền nhau nên rơi vào 2 sub-bank khác nhau. Ở INT8, \{i, jw\} = 8i + jw và 8i chia hết cho 8, nên sub-bank chỉ phụ thuộc jw. Mô hình kiểm đủ mọi ($i_0$, k), (k, $j_0$) và mọi đoạn hàng C.
\end{notebox}

\subsubsection{Chọn bên truy cập}

\paragraph{Mux cổng của mỗi sub-bank (X, b, s)}
\begin{itemize}
\item \sig{dma_sel} = lệnh DMA (ghi \sig{i_dw_*} cho A, B, BIAS; đọc \sig{i_dr_*} cho C) có buffer X, bank b và sub-bank s
\item \sig{cmp_sel} = lệnh phía tính tương ứng (\sig{i_a_rd_en}, \sig{i_b_rd_en}, \sig{i_bias_rd_en}, \sig{i_c_wr_en} với byte enable khác 0 cho s) và \sig{i_comp_bank_X} = b
\item RAM: \sigc{en = dma_sel | cmp_sel}; địa chỉ, dữ liệu, \sig{we}, byte enable lấy từ bên được chọn. Nếu cả hai cùng chọn, DMA thắng
\item Không cần tín hiệu pha: quyền truy cập do bank và lịch của FSM bảo đảm
\end{itemize}

\begin{notebox}[Quyết định baseline]
\textbf{D-12 (trọng tài cổng SRAM).} \sig{en} của mỗi sub-bank RAM = \sig{dma_sel} | \sig{cmp_sel}; khi trùng, DMA thắng. Đồng thời assertion AS-02 (chỉ mô phỏng) khẳng định tình huống này không bao giờ xảy ra khi \sig{i_comp_active} = 1, vì FSM đã cấp bank khác nhau cho hai phía. Lý do: hành vi phải xác định được trong RTL; assertion là cơ chế phát hiện lỗi lập lịch.
\end{notebox}

\subsection{Reset, hiện thực, kiểm chứng}\label{sec:8}

\paragraph{Reset}
\begin{itemize}
\item \sig{rst_n} = 0 hoặc \sig{i_soft_clr}: \sig{o_dr_rvalid} về 0. Thanh ghi chọn byte, bank, sub-bank và nội dung RAM không bị reset
\item \sig{o_dr_rvalid} là flip-flop trạng thái duy nhất của module: \sigc{always_ff @(posedge clk or negedge rst_n)}, nhánh reset và nhánh \sig{i_soft_clr} đều về 0, ngược lại nhận \sig{i_dr_valid}
\end{itemize}

\begin{notebox}[Quyết định baseline]
\textbf{D-21 (kiểu always\_ff).} Flip-flop trạng thái (\sig{o_dr_rvalid}) dùng \sigc{always_ff @(posedge clk or negedge rst_n)}; thanh ghi chỉ chứa dữ liệu (thanh ghi chọn byte, bank, sub-bank; RAM; thanh ghi đầu ra RAM) dùng \sigc{always_ff @(posedge clk)} và không có giá trị reset. \sig{soft_clr} là xóa đồng bộ, không đi vào chân reset.
\end{notebox}

\paragraph{Hiện thực}
\begin{itemize}
\item RAM viết theo mẫu \sig{gemm_spm_ram} để công cụ suy ra block RAM hoặc LUTRAM có đọc đồng bộ. Không được dùng RAM đọc bất đồng bộ cho A, B, C, BIAS
\item Không có mux tổ hợp từ \sig{*_rd_en} tới dữ liệu ra; đường ra là: thanh ghi đầu ra RAM → mux bank → mux byte/lane. Thanh ghi chọn bank, byte (k[1:0]) và sub-bank được nạp ở mỗi chu kỳ từ các cổng địa chỉ
\item Mỗi buffer trong file riêng; NB và số sub-bank dựng bằng \sig{generate}
\end{itemize}

\paragraph{Assertion bắt buộc (SVA)}
\begin{itemize}
\item \textbf{AS-01}: \sig{i_dw_valid} → \sig{i_dw_buf} ≠ 3 ∧ địa chỉ trong miền của buffer (A, B < 256; BIAS < 32)
\item \textbf{AS-02}: khi \sig{i_comp_active} = 1, không sub-bank nào có \sig{dma_sel} ∧ \sig{cmp_sel} cùng chu kỳ; ở mức bank: \sig{i_dw_valid} ∧ \sig{i_dw_buf} ∈ \{A, B\} → \sig{i_dw_bank} ≠ \sig{i_comp_bank_X} với X = \sig{i_dw_buf}; \sig{i_dw_valid} ∧ \sig{i_dw_buf} = BIAS ∧ \sig{i_bias_rd_en} → \sig{i_dw_bank} ≠ \sig{i_comp_bank_bias}; \sig{i_dr_valid} → \sig{i_dr_bank} ≠ \sig{i_comp_bank_c}. Thành phần BIAS chỉ áp khi phía tính đang đọc bias, vì khi \sig{POST_EN} = 0 Main FSM được phép nạp BIAS cho job sau vào bank trùng \sig{i_comp_bank_bias} (không phải xung đột). Ràng buộc lập lịch đầy đủ cho BIAS nằm ở AS-04 của Main FSM. Đây là assertion của quyết định D-12: RTL vẫn xác định (DMA thắng) nếu vi phạm, assertion phát hiện lỗi lập lịch
\item \textbf{AS-03}: \sig{i_a_rd_en} ∨ \sig{i_b_rd_en} ∨ \sig{i_bias_rd_en} ∨ \sig{i_c_wr_en} → \sig{i_comp_active}
\item \textbf{AS-04}: \sig{i_a_rd_row}, \sig{i_b_rd_col}, \sig{i_bias_rd_col}, \sig{i_c_wr_col} là bội của 8 khi có lệnh
\item \textbf{AS-05}: \sig{o_dw_ready} = \sig{o_dr_ready} = 1 ở mọi chu kỳ; \sig{o_dr_rvalid} bằng \sig{i_dr_valid} của chu kỳ trước
\item \textbf{AS-06}: \sig{i_c_wr_en} ∧ \sig{i_c_wr_fmt} = 1 → \sig{i_c_wr_data[255:64]} = 0
\end{itemize}

\paragraph{Kiểm tra chức năng}
\begin{itemize}
\item Ghi A, B, BIAS qua cổng DMA rồi đọc qua cổng array/VPU: mọi (i$_0$, k), (k, $j_0$), $j_0$ khớp giá trị đã ghi, đúng 1 chu kỳ sau \sig{*_rd_en}
\item Ghi C INT32 và INT8 qua cổng VPU với mask ngẫu nhiên rồi đọc qua cổng DMA: byte/word có mask = 0 giữ nguyên giá trị cũ, phần tử có mask = 1 khớp
\item Phía tính và DMA truy cập đồng thời hai bank khác nhau của cùng buffer: không sai lệch dữ liệu ở cả hai phía
\item Số sub-bank bị chạm trong mỗi truy cập song song bằng đúng bảng ở mục điều kiện phục vụ đồng thời, không sub-bank nào bị chạm hai lần
\end{itemize}
