% ===================== 1 =====================
\section{Chức năng}\label{sec:1}

\begin{keypoint}
Scratchpad giữ bốn buffer logic A, B, C, BIAS, mỗi buffer có NB bank. Phía tính (array, VPU) cần đọc hoặc ghi P phần tử trong một chu kỳ, nên mỗi bank được chia thành nhiều \textbf{sub-bank} một cổng sao cho mỗi truy cập song song chạm các sub-bank khác nhau. Phía DMA đọc hoặc ghi một word mỗi chu kỳ. Hai phía dùng hai bank khác nhau (ping-pong) hoặc hai pha khác nhau (không ping-pong), nên không bao giờ tranh nhau một sub-bank.
\end{keypoint}

\begin{reqtable}{Chức năng}{tab:1-1}
SRM-F01 & Mỗi bank có dung lượng đúng SPM-01: A 256 word, B 256 word, C 1024 word, BIAS 32 word; tổng NB × 1568 word (NB = 2: 12,25 KiB) \\
SRM-F02 & Mọi cổng đọc có độ trễ cố định 1 chu kỳ; mọi lệnh ghi có hiệu lực ở cạnh clock cuối chu kỳ có lệnh; không có backpressure (SPM-04, SPM-05) \\
SRM-F03 & Mỗi sub-bank vật lý là RAM một cổng đồng bộ. Bank b của buffer X chỉ bị truy cập bởi một phía trong một chu kỳ (PP-07); hai bank khác nhau được truy cập độc lập cùng chu kỳ \\
SRM-F04 & Nội dung không được khởi tạo bởi \sig{rst_n} hay \sig{soft_clr} (SPM-06) \\
\end{reqtable}

\subsection{Truy cập phía tính}
\begin{reqtable}{Cổng phía tính (b = \sigc{i_comp_bank_*})}{tab:5-3}
MAP-01 & Đọc A ($i_0$, k): sub-bank r (0 ≤ r < P) của A[b] đọc địa chỉ \{$i_0$[4:LP], k[4:2]\}. Lane r = byte k[1:0] của sub-bank r; k[1:0] được chốt để chọn byte ở chu kỳ sau \\
MAP-02 & Đọc B (k, $j_0$): sub-bank s (0 ≤ s < Q) của B[b] đọc \{k, ($j_0$ » 2) » log$_2$Q\}. Lane c = byte (c mod 4) của sub-bank c/4 \\
MAP-03 & Đọc BIAS ($j_0$): sub-bank c của BIAS[b] đọc $j_0$ » LP; lane c = sub-bank c \\
MAP-04 & Ghi C INT32 (i, $j_0$): lane c ghi sub-bank c của C[b] tại (\{i, $j_0$\} » LP), cả 4 byte, nếu \sig{mask[c]} \\
MAP-05 & Ghi C INT8 (i, $j_0$): word thứ w (0 ≤ w < Q) của đoạn hàng có địa chỉ a = \{i, $j_0$/4 + w\}, nằm ở sub-bank s = ($j_0$/4 + w) mod P, địa chỉ a » LP. Sub-bank s nhận word w = (s − $j_0$/4) mod P nếu w < Q; byte e của nó = \sig{i_c_wr_data[8(4w+e) +: 8]}, byte enable e = \sig{mask[4w+e]} \\
\end{reqtable}

\subsection{Truy cập phía DMA}
\begin{reqtable}{Cổng DMA}{tab:5-4}
MAP-06 & Ghi \sig{i_dw_*}: sub-bank và địa chỉ cục bộ theo bảng~\ref{tab:5-1} với a = \sig{i_dw_addr}, trong buffer \sig{i_dw_buf}, bank \sig{i_dw_bank}; cả 4 byte \\
MAP-07 & Đọc \sig{i_dr_*}: sub-bank (a mod P) của C[\sig{i_dr_bank}], địa chỉ a » LP. Số sub-bank và bank được chốt để chọn dữ liệu ra ở chu kỳ sau \\
\end{reqtable}

\begin{notebox}[Vì sao không bao giờ xung đột]
$i_0$ và $j_0$ là bội của P, nên P hàng của một đọc A rơi vào P sub-bank khác nhau, P cột của một đọc BIAS hoặc ghi C INT32 cũng vậy; Q word của một đọc B hoặc ghi C INT8 nằm liền nhau nên rơi vào Q sub-bank khác nhau. Ở INT8, \{i, jw\} = 8i + jw và 8i chia hết cho P, nên sub-bank chỉ phụ thuộc jw. Mô hình kiểm đủ mọi (i$_0$, k), (k, j$_0$), mọi đoạn hàng C, với P = 4, 8.
\end{notebox}

\subsection{Chọn bên truy cập}
\begin{reqtable}{Mux cổng của mỗi sub-bank (X, b, s)}{tab:5-5}
MUX-01 & \sig{dma_sel} = lệnh DMA (ghi \sig{i_dw_*} cho A, B, BIAS; đọc \sig{i_dr_*} cho C) có buffer X, bank b và sub-bank s \\
MUX-02 & \sig{cmp_sel} = lệnh phía tính tương ứng (\sig{i_a_rd_en}, \sig{i_b_rd_en}, \sig{i_bias_rd_en}, \sig{i_c_wr_en} với byte enable khác 0 cho s) và \sig{i_comp_bank_X} = b \\
MUX-03 & RAM: \sig{en = dma_sel ∨ cmp_sel}; địa chỉ, dữ liệu, byte enable lấy từ bên được chọn. Nếu cả hai (vi phạm, AS-02), DMA thắng \\
MUX-04 & Không cần tín hiệu pha: quyền truy cập do bank và lịch của FSM bảo đảm (SB-01, SB-02 spec FSM) \\
\end{reqtable}

\begin{Verbatim}[frame=leftline,framerule=2pt,rulecolor=\color{brand},xleftmargin=6pt,samepage=true]
// gemm_spm_ram: RAM một cổng, đọc đồng bộ (tham khảo)
logic [31:0] mem [DEPTH];
always_ff @(posedge clk)
  if (i_en) begin
    if (i_we) begin
      for (int e = 0; e < 4; e++)
        if (!BE || i_be[e]) mem[i_addr][8*e +: 8] <= i_wdata[8*e +: 8];
    end else
      o_rdata <= mem[i_addr];
  end
\end{Verbatim}

\section{Reset, hiện thực, kiểm chứng}\label{sec:8}
\begin{reqtable}{Reset}{tab:8-1}
RST-S01 & \sig{rst_n} = 0 hoặc \sig{i_soft_clr}: \sig{o_dr_rvalid} về 0. Thanh ghi chọn byte, bank, sub-bank và nội dung RAM không bị reset \\
\end{reqtable}

\begin{reqtable}{Hiện thực}{tab:8-2}
IMP-01 & RAM viết theo mẫu \sig{gemm_spm_ram} để công cụ suy ra block RAM hoặc LUTRAM có đọc đồng bộ. Không được dùng RAM đọc bất đồng bộ cho A, B, C (sai timing SRM-F02) \\
IMP-02 & Không có mux tổ hợp từ \sig{*_rd_en} tới dữ liệu ra; đường ra là: thanh ghi đầu ra RAM → mux bank → mux byte/lane \\
IMP-03 & Mỗi buffer trong file riêng; NB và số sub-bank dựng bằng \sig{generate} \\
\end{reqtable}

\begin{reqtable}{Assertion bắt buộc (SVA)}{tab:8-3}
AS-01 & \sig{i_dw_valid} → \sig{i_dw_buf} ≠ 3 ∧ địa chỉ trong miền của buffer (A, B < 256; BIAS < 32) \\
AS-02 & NB = 2: \sig{i_dw_valid} ∧ \sig{i_comp_active} → \sig{i_dw_bank} ≠ \sig{i_comp_bank_X} với X = \sig{i_dw_buf}; và \sig{i_dr_valid} ∧ \sig{i_comp_active} → \sig{i_dr_bank} ≠ \sig{i_comp_bank_c}. NB = 1: \sig{i_dw_valid} ∨ \sig{i_dr_valid} → ¬\sig{i_comp_active} (OWN-02) \\
AS-03 & \sig{i_a_rd_en} ∨ \sig{i_b_rd_en} ∨ \sig{i_bias_rd_en} ∨ \sig{i_c_wr_en} → \sig{i_comp_active} \\
AS-04 & Không sub-bank nào có \sig{dma_sel} ∧ \sig{cmp_sel} cùng chu kỳ \\
AS-05 & \sig{i_a_rd_row}, \sig{i_b_rd_col}, \sig{i_bias_rd_col}, \sig{i_c_wr_col} là bội của P khi có lệnh \\
\end{reqtable}
