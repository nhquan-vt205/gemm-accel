\section{Port top-level}\label{sec:6}

\begin{keypoint}
Port của \sig{gemm_accel_top} \textbf{đã khóa}: clock, reset, một cổng AXI4-Lite slave (17 tín hiệu) và một cổng AXI4-Full master (25 tín hiệu), tổng 44 port. Mọi thay đổi về sau, kể cả phần bổ sung ở mục~\ref{sec:7}, chỉ được thay đổi port nội bộ giữa các module, không được thêm, bớt hay đổi port ở bảng~\ref{tab:6-2} và bảng~\ref{tab:6-3}.
\end{keypoint}

\subsection{Tham số}\label{sec:6.1}

\begin{spectable}{L{0.27}L{0.15}L{0.58}}{Tham số của \sigc{gemm_accel_top}}{tab:6-1}{\textbf{Tham số} & \textbf{Mặc định} & \textbf{Ý nghĩa}}
\sig{PE_ARRAY_SIZE} & 8 & P; chỉ nhận 4 hoặc 8 \\
\sig{MAX_BURST_BEATS} & 16 & Số beat tối đa mỗi burst, 1 … 256 \\
\sig{HAS_CHAIN} & 0 & 1 nếu hiện thực chế độ chuỗi descriptor (mục~\ref{sec:7.2}) \\
\sig{HAS_PINGPONG} & 0 & 1 nếu hiện thực scratchpad ping-pong (mục~\ref{sec:7.3}); đòi hỏi \sig{HAS_CHAIN} = 1 \\
\end{spectable}
